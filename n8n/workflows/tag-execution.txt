const b = ($('QuickReply Webhook').first().json.body) || {};

// Match "Is Real User Message?" EXACTLY: a real message is one whose payload carries a
// USER_* _type.
// HISTORY, and read this before "simplifying" it again. This once tested `b.payload._type`,
// was changed to `!b.payload` because the workflow's own PINNED SAMPLE body had a payload
// with text and no _type - i.e. it was weakened to accommodate a test fixture, not real
// traffic. Real traffic proved the opposite: of 1203 webhook requests logged on 10 Sep 2026,
// 139 were genuine messages (all with payload._type USER_TEXT / USER_BUTTON_REPLY, plus
// msg_time, messageBy USER, no event) and 1064 were SENT/DELIVERED/READ status callbacks -
// which DO carry a payload with the message text, and no _type. Not one body had both.
// So `!b.payload` labelled all 1064 callbacks as MESSAGE, and the identical test in
// "Is Real User Message?" let them start a full AI run, which is how the bot ended up
// answering its own outbound messages three times over.
// The pinned sample now carries "_type": "USER_TEXT" so the fixture matches reality.
const isStatus = !b.payload || !b.payload._type;

const tags = {
  kind: isStatus ? 'STATUS' : 'MESSAGE',
  phone: String(b.phone || ''),
};
if (isStatus) {
  tags.event = String(b.event || '');
  tags.by = String(b.messageBy || '');
} else {
  tags.text = String((b.payload && b.payload.text) || '').slice(0, 60);
  // Kept so a media/button message is identifiable in the list without opening it.
  tags.ptype = String((b.payload && b.payload._type) || 'none');
}

// customData is a PAID n8n feature and is stored on PRODUCTION executions only - a manual
// run never keeps it. The old code swallowed the failure with a bare comment, so "no tags
// in the executions list" was indistinguishable from "this plan does not support tags".
// Log it instead: one line in the execution log tells you which of the two it is.
// Limits: 10 keys per execution, key <= 50 chars, value <= 512 chars.
try {
  for (const k of Object.keys(tags)) $execution.customData.set(k, tags[k]);
} catch (e) {
  console.log('[Tag Execution] customData NOT saved - ' + e.message
    + ' (paid feature, and production executions only). Tags attempted: ' + JSON.stringify(tags));
}

return $input.all();
