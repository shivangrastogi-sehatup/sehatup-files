// ---- Sanitize + strip ANY model-generated intro (kept: time greetings, titles) ----
function sanitizeReply(t) {
  if (!t) return '';
  let s = String(t);
  s = s.replace(/\b(good\s*(morning|afternoon|evening|noon)|gud\s*(mrng|morning|evening|evng)|shubh\s*(prabhat|sandhya)|subh\s*prabhat)\b[\s,!.-]*/gi, '');
  // Titles - the customer's gender is unknown, so none of these may ever survive.
  // Covers bare "mam", curly apostrophes and a trailing "-ji" ("mam ji" -> "ji").
  // The optional ji is inside its own group so a plain "mam aapko" does not eat the space.
  s = s.replace(/\b(ma['\u2019`]?am|m['\u2019`]am|mam|madam|mdm|mem(\s*(saab|sahab|saheb|sahib))?)(\s*(ji|jee))?\b/gi, 'ji');
  // "sirf" is safe (no word boundary); "sir dard" is a headache, not an address.
  s = s.replace(/\bsirs?(\s*(ji|jee))?\b(?!\s*(dard|dukh|dukhna|pain|ache|ghum|ghoom|chakkar|bhari|heavy|me\b|mein\b))/gi, 'ji');
  s = s.replace(/\b(bhai|bhaiya|bhaiyya|bhaisahab|bro|brother|behen|bahen|behn|didi|bhabhi)(\s*(ji|jee))?\b/gi, 'ji');
  // Devanagari equivalents: madam / maim / mem / sarji / bhaiya / didi / bahan-ji -> "ji".
  // Written as \u escapes so copy-paste can never corrupt them. Bare "sar" (U+0938 U+0930)
  // also means head, so it is deliberately NOT matched.
  s = s.replace(/(\u092E\u0948\u0921\u092E|\u092E\u0948\u092E|\u092E\u0947\u092E|\u092E\u0947\u092E\u0938\u093E\u0939\u092C|\u0938\u0930\s*\u091C\u0940|\u0938\u0930\u091C\u0940|\u092D\u0948\u092F\u093E|\u092D\u093E\u0908\u0938\u093E\u0939\u092C|\u092D\u093E\u0908\s*\u0938\u093E\u0939\u092C|\u0926\u0940\u0926\u0940|\u092C\u0939\u0928\s*\u091C\u0940)/g, '\u091C\u0940');
  // Speaker label leaked from the fine-tune's training format ("Ananya:- ...").
  // Multiline: the model puts it on continuation lines too, not just the first.
  s = s.replace(/^[\s>*-]*(ananya|agent|bot|assistant)\s*(ji)?\s*[:\-\u2013\u2014]+\s*/gim, '');
  s = s.replace(/hello\s*ji[\s,!.]*/gi, '');
  // Self-introductions are stripped ONLY when the customer did not ask who she is. If they
  // did, this is the answer to their question.
  if (!_askedIdentity) {
    // Sentence-level first, and word-order-independent. The two patterns below both hard-
    // require an "m" before "ananya" (/\bma?i?n?\s*ananya/) AND require "ananya" to sit
    // immediately before "baat kar" - so the two variants the model actually sent
    // mid-conversation, "Ananya baat kar rahi hu SehatUP se" and "Mai SehatUP se baat kar
    // rahi hu", matched neither and reached the customer as a re-introduction.
    const QR_INTRO_SENT = /(\bananya\b[^.!?\n]{0,40}\b(baat\s*kar|bol\s*rah|hu\b|hoon\b)|\bsehat\s*up\s*se\b[^.!?\n]{0,30}\b(baat\s*kar|bol\s*rah))/i;
    s = s.split(/(?<=[.!?])\s+|\n+/).filter(x => !QR_INTRO_SENT.test(x)).join(' ');
    s = s.replace(/\bma?i?n?\s*ananya\s+baat\s+kar\s+rah[iy]\s*hu?n?\s*,?\s*(sehat\s*up\s*se)?[\s,.!]*/gi, '');
    // "SehatUP se" is swallowed with the intro, not left behind: without it
    // "Ji, main Ananya hu SehatUP se." sanitized to the fragment "Ji, SehatUP se.".
    s = s.replace(/\bma?i?n?\s*ananya\s+(hu|hoon)\b\s*,?\s*(sehat\s*up\s*se)?[\s,.!]*/gi, '');
  }
  s = s.replace(/\b(ji)(\s*,?\s*ji\b)+/gi, 'ji');
  s = s.replace(/(\u091C\u0940)(\s*,?\s*\u091C\u0940)+/g, '\u091C\u0940');
  s = s.replace(/[ \t]{2,}/g, ' ').replace(/\n{3,}/g, '\n\n');
  // Trailing debris too, not just leading: stripping the intro out of "Ji, mai Ananya baat
  // kar rahi hu SehatUP se." leaves "Ji," - a dangling comma the old trim kept.
  s = s.replace(/^[\s,.:;!-]+/, '').replace(/[ \t]+([,.!?])/g, '$1').replace(/[\s,;:-]+$/, '').trim();
  return s;
}
// Kept for the role-reversal guard below, which has to see what the model ACTUALLY wrote:
// sanitizeReply strips time greetings and speaker labels, and "Good morning Ananya," - the
// clearest evidence the model was writing as the customer - survives only here.
const _rawModelText = String(aiText || '');
aiText = sanitizeReply(aiText);

// ---- DID SANITIZING EAT THE WHOLE REPLY? ----
// The intro-stripping above can consume the model's ENTIRE output. When the model answers
// with nothing but its own introduction ("Ji, mai Ananya baat kar rahi hu SehatUP se."),
// what survives is the filler "Ji" - and the greet-once block at the bottom then prepends
// the workflow's own intro to it, producing the message customers actually received:
//     "Hello ji, mai Ananya baat kar rahi hu SehatUP se. ji"
// A leftover that is pure filler means the model produced no reply, so treat it as empty:
// on first contact the intro block supplies the proper opener instead.
const QR_FILLER_RE = /^(ji|jee|g|ok|okay|okey|k|hmm+|hm|acha|accha|achha|theek|thik|theek hai|thik hai|haan|han|ha|yes|sure|right|ji haan|ji ok|noted|welcome|most welcome)[\s.,!]*$/i;

// A reply can be content-free without being one of the exact words above. QR_FILLER_RE is
// anchored, so it recognises a bare "ji" and nothing else - and production sent
// "ji yah drops hai" TWICE IN A ROW to someone asking which drops they had been offered.
// Four words, no product name, no price, no link, no question: it is the same non-answer,
// it just did not look like one to an anchored regex.
// A reply is treated as substantive if it asks something, quotes a price or sends a link -
// otherwise anything this short is the model stalling.
const QR_NONANSWER_MAX_WORDS = 5;
function qrIsNonAnswer(t) {
  const s = String(t || '').trim();
  if (!s) return true;
  if (QR_FILLER_RE.test(s)) return true;
  if (/\?/.test(s)) return false;
  // No rupee SIGN here, deliberately: this file must stay pure ASCII (see replyHasPrice
  // below for why), and every price this workflow writes goes through _rs(), which emits
  // "Rs 849". A reply carrying the symbol instead came from the model, and the price guard
  // below is what handles that case.
  if (/(rs\.?\s*\d{2,}|\b\d{3,}\s*(rupees|rupaye|rupay)\b|sehatup\.com|https?:\/\/)/i.test(s)) return false;
  return s.split(/\s+/).filter(Boolean).length <= QR_NONANSWER_MAX_WORDS;
}
if (!greetedBefore && QR_FILLER_RE.test(String(aiText || '').trim())) {
  console.log('[Sanitize] first-contact reply was intro-only - blanked so the opener is used | was=' + JSON.stringify(aiText));
  aiText = '';
}

// What the MODEL said, before any guard rewrote it. The safety-claim guard at the bottom reads
// this as well as the final text: a claim that another guard happened to overwrite still needs
// to be counted, because "the model tried to promise a medicine has no side effects" is a fact
// about the model worth having in the execution tags even when the customer never saw it.
const _modelText = String(aiText || '');

// ---- ROLE-REVERSAL GUARD (runs before every other guard) ----
// It is deliberately NOT in _answered(). Its job is to make the model's output unusable, not
// to own the reply: it drops a safe, topic-neutral floor in place, and any later guard that
// builds a BETTER answer from the customer's own question - the health-score link, the dose
// handoff, the price reply, the out-of-hours slot correction - is still allowed to override
// it. The guards that only inspect the model's text (claim, promise, substance) then read
// this safe replacement and correctly stay silent.
// 2026-08-05, execution 19013. A first-contact customer asked "Hello! Can I get more info for
// PCOD/PCOS?" then "K", and was sent back:
//     "Ananya, Yes, I need to know about PCOS and its treatment. I'm suffering from PCOS,
//      since 2017. I have hair fall, acne, and pigmentation. I used medications before but
//      nothing helped. K"
// The model wrote the CUSTOMER'S next turn instead of Ananya's. It addressed Ananya, spoke in
// the first person as the patient, and copied her own trailing "K". The bot therefore invented
// a medical history - PCOS since 2017, hair fall, acne, pigmentation, failed prior medication -
// and sent it to the patient as though she had said it. It is also poisoned training data: it
// is stored with senderKind 'AI' and reads as a genuine patient turn.
//
// All thirteen other guards returned false, and none of them was broken. Every one asks WHAT
// the reply claims - a dose, a price, a score, a diagnosis, a callback promise. Not one asks
// WHO IS SPEAKING, and a reply written in the customer's voice claims nothing forbidden.
//
// Why the model does it: the system prompt is ~26,000 characters whose largest block is
// EXAMPLES written as two-sided dialogue ("Cust "..." -> "..."", sixteen of them). On first
// contact there is NO assistant turn in the history to anchor the role, so a transcript-shaped
// prompt gets CONTINUED rather than answered. That makes it likeliest exactly when the stakes
// are highest - a stranger's opening message, where there is no history to fall back on.
// (avgLogprobs was -1.79 on that call, against roughly -0.5 for a confident reply.)
let roleGuard = false;
{
  // Both texts: signals 1 and 2 can appear in either, since sanitizeReply removes the leading
  // "Good morning" and any "Ananya:" speaker label before the guards ever run.
  const _rt = _rawModelText + '\n' + String(aiText || '');

  // 1. It addresses Ananya. She never addresses herself; a self-introduction ("mai Ananya
  //    baat kar rahi hu") has no comma or colon after the name, so it cannot match here.
  const _vocativeAnanya =
       /(^|\n)\s*((hi|hello|hey|namaste|dear|good\s*(morning|afternoon|evening))\b[\s,]*)?ananya\s*[,:]/i.test(_rt)
    || /\b(to|dear|thanks|thank\s*you|ok|okay)\s+ananya\b/i.test(_rt);

  // 2. It speaks as the patient. Ananya's own first person is always about helping ("main
  //    check kara deti hu"), never about having a symptom, so the condition/symptom word is
  //    what makes this safe.
  const _SYMPTOMS = '(pcod|pcos|thyroid|diabetes|sugar|bp|acne|pigmentation|hair\\s*fall|hairfall|infertility|anxiety|stress|dard|pain|problem|issue)';
  const _patientVoice =
       new RegExp('\\b(i\\s*am|i\\s*m|im)\\s+(suffering|facing|dealing)\\b', 'i').test(_rt)
    || new RegExp('\\bi\\s+(have|had|ve\\s*got|got)\\s+(been\\s+)?' + _SYMPTOMS, 'i').test(_rt)
    || new RegExp('\\bi\\s+(used|tried|took)\\s+(medication|medicine|tablet|tablets|treatment|dawa)', 'i').test(_rt)
    || new RegExp('\\bmujhe\\s+' + _SYMPTOMS + '\\b[^.!?]{0,14}\\b(hai|h)\\b', 'i').test(_rt)
    || new RegExp('\\bmer[ia]\\s+' + _SYMPTOMS + '\\b[^.!?]{0,14}\\b(hai|h)\\b', 'i').test(_rt);

  // 3. It ends by echoing one of the customer's own messages verbatim - the "K" above. A real
  //    reply never closes by quoting back a whole line the customer just sent.
  const _custLines = String(newMsgText || '').split(/\n+/).map(s => s.trim()).filter(Boolean);
  const _replyLines = _rawModelText.split(/\n+/).map(s => s.trim()).filter(Boolean);
  const _lastLine = (_replyLines[_replyLines.length - 1] || '').toLowerCase();
  const _echoesCustomer = _replyLines.length > 1 && !!_lastLine
    && _custLines.some(c => c.toLowerCase() === _lastLine);

  // 4. The whole reply IS the customer's message, handed back. 2026-08-08, exec 25843:
  //      customer -> "Mujhe Vaji Bati or Kern Drops Chahiye"
  //      model    -> "Good morning sir mujhe Vaji Bati or Kern Drops chahiye"
  //    and what went out, after sanitizing and the first-contact intro, was
  //      "Hello ji, mai Ananya baat kar rahi hu SehatUP se. ji mujhe Vaji Bati or Kern
  //       Drops chahiye"
  //    - Ananya introducing herself and then ordering products from the customer.
  //
  //    All three signals above missed it: it never names Ananya (it says "sir"), it carries
  //    no SYMPTOM word so patient_voice could not fire on a PRODUCT request, and signal 3
  //    only inspects the last line of a MULTI-line reply while this was a single line.
  //
  //    Compared on token overlap, not string equality, so a "Good morning sir" prefix and a
  //    change of case cannot hide it.
  const _norm = (s) => String(s || '').toLowerCase().replace(/[^a-z0-9\s]+/g, ' ').replace(/\s+/g, ' ').trim();
  const _replyTok = _norm(_rawModelText)
    .replace(/\b(good\s*(morning|afternoon|evening)|hello|hi|hey|namaste|ji|sir|mam|madam|ok|okay)\b/g, ' ')
    .split(/\s+/).filter(w => w.length > 1);
  const _custTok = new Set(_norm(newMsgText).split(/\s+/).filter(Boolean));
  const _overlap = _replyTok.length ? _replyTok.filter(w => _custTok.has(w)).length / _replyTok.length : 0;
  // A reply that sends a link, quotes a price or asks something is doing real work even when
  // it reuses the customer's words ("Aap Vaji Bati ki baat kar rahe hain? Ye raha link: ...").
  const _addsSomething = /(https?:\/\/|sehatup\.com|rs\.?\s*\d{2,}|\?)/i.test(_rawModelText);
  const _parrotsCustomer = _replyTok.length >= 3 && _overlap >= 0.8 && !_addsSomething;

  if (_vocativeAnanya || _patientVoice || _echoesCustomer || _parrotsCustomer) {
    roleGuard = _vocativeAnanya ? 'addressed_ananya'
      : (_patientVoice ? 'patient_voice'
      : (_echoesCustomer ? 'echoed_customer' : 'parroted_customer'));
    // Topic-neutral on purpose: this fires on any subject, so it cannot assume what was asked.
    // It promises nothing (BOOKING POLICY rule 4) and states no condition (CLAIM POLICY).
    aiText = 'Ji bilkul, main aapki help karti hu. Aap thoda detail me bata dijiye - problem kya hai aur kab se hai? '
      + 'Free consultation me doctor aapko sahi plan bata denge, consultation aur diet plan dono free hain.';
    console.log('[Role Guard] BLOCKED ' + roleGuard + ' - model replied as the CUSTOMER'
      + ' | was=' + JSON.stringify(_rawModelText.slice(0, 120)));
  }
}

// ---- MEDICAL CLAIM GUARD (outranks every guard except the role-reversal guard above) ----
// Asked "I want to Check My Free PCOD Health Score", the model replied:
//     "ji / your score is 7 / you have a PCOD / and you need to consultation for it"
// A fabricated score and a fabricated diagnosis. There is no health-score integration at
// all, so the number was invented outright. This is the worst output the system can
// produce - worse than a dose - because the customer has no way to know it is fiction, and
// it is a medical claim made to a patient. It runs before everything else: whatever else is
// true about the message, a reply containing an invented score or diagnosis must not go out.
//
// Echoing the customer's own words is NOT diagnosing. If they said "mera PCOD hai", Ananya
// saying "aapko PCOD hai" is ordinary conversation, so both checks are suppressed when the
// customer themselves raised the condition or the number.
const _qLower = String(newMsgText || '').toLowerCase();
const _CONDITIONS = '(pcod|pcos|thyroid|hypothyroid|hyperthyroid|diabetes|sugar|bp|blood\\s*pressure|infertility|anemia|anaemia|cancer|hormonal\\s*imbalance)';
const _custRaisedCondition = new RegExp(_CONDITIONS, 'i').test(_qLower);
const _custGaveNumber = /\b(score|skor)\b[^.!?]{0,20}\d/i.test(_qLower);

const gaveScore = !_custGaveNumber && (
     /\b(score|skor)\b[^.!?]{0,24}\b\d{1,3}\b/i.test(aiText)
  || /\b\d{1,3}\s*(\/|out\s*of)\s*(10|100)\b/i.test(aiText)
);

// An assertion, not a question. The persona REQUIRES Ananya to ask "aapko thyroid, sugar ya
// BP ki koi problem hai?" (safety gating, rule 3) - that must survive untouched, so any
// sentence that is a question or a conditional is skipped.
function qrAssertsCondition(t) {
  const RE = new RegExp('\\b(aapko|apko|aap\\s*ko|aapka|apka|you\\s*have|you\\s*are\\s*having|u\\s*have)\\b[^.!?]{0,26}\\b' + _CONDITIONS + '\\b', 'i');
  const sentences = String(t || '').split(/(?<=[.!?])\s+|\n+/);
  for (const s of sentences) {
    if (/\?/.test(s)) continue;
    if (/\bkya\b/i.test(s)) continue;
    if (/\b(agar|if|jab|yadi|shayad|ho\s*sakta)\b/i.test(s)) continue;
    if (RE.test(s)) return true;
  }
  return false;
}
const gaveDiagnosis = !_custRaisedCondition && qrAssertsCondition(aiText);

// ---- WHATSAPP PROFILE NAME GUARD ----
// The reply "PCOD Health Score Check for My Love My Papa" used the customer's WhatsApp
// PROFILE name. Profile names are whatever someone set for themselves - "My Love My Papa",
// emoji, a shop name, a phone number - and are not the person's name. Reading one back makes
// the bot sound broken. A plausible-looking name is left alone; anything else is scrubbed
// out of the reply entirely.
const _rawName = String(name || '').trim();
const _nameIsPlausible = /^[A-Za-z][A-Za-z.'-]{1,20}(\s+[A-Za-z][A-Za-z.'-]{1,20}){0,2}$/.test(_rawName)
  && !/(love|papa|mumm?y|\bmom\b|dad+y?|jaan|baby|king|queen|\bji\b|\bsir\b|madam|boss|bhai|didi|angel|princess|hero|star|customer|unknown)/i.test(_rawName);
let nameGuard = false;
// Three safeguards, all of them earned:
//  1. MIN LENGTH 3 - a display name of "a" stripped every letter a from the reply, turning
//     "aap tension mat lijiye" into "p tension mt lijiye". Reached real customers.
//  2. WORD BOUNDARIES - otherwise "Ram" would gut "program" and "Ana" would gut "banana".
//  3. LENGTH SANITY - if the strip would remove more than 40% of the reply, something is
//     wrong with the name and the original is kept. A mangled reply is worse than a stray name.
const NAME_MIN_STRIP = 3;
if (_rawName && !_nameIsPlausible && _rawName.length >= NAME_MIN_STRIP) {
  const _esc = _rawName.replace(/[.*+?^${}()|[\]\\]/g, '\\$&');
  const _re = new RegExp('\\b' + _esc + '\\b', 'ig');
  const _hits = (aiText.match(_re) || []).length;
  // A "name" that appears 4+ times is a common word (someone called "Hai" or "Hum"),
  // not a name. Refuse rather than shred the sentence.
  if (_hits > 0 && _hits <= 3) {
    const _stripped = aiText.replace(_re, '')
      .replace(/\s+for\s*$/i, '').replace(/\s*,+\s*,/g, ',')
      .replace(/\s{2,}/g, ' ').replace(/\s+([,.!?])/g, '$1').trim();
    if (_stripped && _stripped.length >= aiText.length * 0.4) {
      aiText = _stripped;
      nameGuard = true;
      console.log('[Name Guard] stripped the WhatsApp profile name | name=' + JSON.stringify(_rawName));
    } else {
      console.log('[Name Guard] SKIPPED - stripping ' + JSON.stringify(_rawName) + ' (' + _hits + ' hits) would have destroyed the reply');
    }
  }
}

// ---- HEALTH REPORT RESPONDER ----
// "Hi I need help regarding my health report" almost always means one thing: they finished
// the questionnaire but never pressed "Get My Report on WhatsApp" on the final page, so it
// was never delivered. The report exists regardless - the submission is written on

// Pull a clock time out of Hinglish. Requires an explicit time marker (baje / am / pm /
// o'clock) so "5 months se" and "10 tablets" can never be read as a slot.
function qrParseSlot(text) {
  const s = String(text || '').toLowerCase();
  const re = /(\d{1,2})(?:[:.](\d{2}))?\s*(a\.?m\.?|p\.?m\.?|baje|bje|bajey|bajay|baze|bajke|o'?\s?clock)/g;
  let m, best = null;
  while ((m = re.exec(s)) !== null) {
    let h = parseInt(m[1], 10);
    const min = m[2] ? parseInt(m[2], 10) : 0;
    if (h < 1 || h > 24 || min > 59) continue;
    const marker = m[3] || '';
    const around = s.slice(Math.max(0, m.index - 26), m.index + m[0].length + 26);
    let pm = null;
    if (/^p/.test(marker)) pm = true;
    else if (/^a/.test(marker)) pm = false;
    else if (/(raat|raat\s*ko|\brat\b|night|shaam|\bsham\b|evening|dopahar|dopeher|afternoon)/.test(around)) pm = true;
    else if (/(subah|savere|sawere|\bsubh\b|morning)/.test(around)) pm = false;
    // No period word: in Indian usage a bare "4 baje" means 4 PM and "10 baje" means 10 AM.
    else pm = (h >= 1 && h <= 6);
    if (pm && h < 12) h += 12;
    if (!pm && h === 12) h = 0;
    if (h === 24) h = 0;
    best = h * 60 + min;   // last time mentioned wins
  }
  return best;
}

// In booking context a bare number IS a time, so the marker requirement is relaxed. Outside
// that context it stays strict, otherwise "5 months se" would read as 5 o'clock.
function qrParseBareTime(text) {
  const m = String(text || '').trim().match(/^\s*(\d{1,2})(?:[:.]?(\d{2}))?\s*$/);
  if (!m) return null;
  let h = parseInt(m[1], 10);
  const min = m[2] ? parseInt(m[2], 10) : 0;
  if (h < 1 || h > 24 || min > 59) return null;
  if (h >= 1 && h <= 6) h += 12;          // "4" or "6:30" means the afternoon, as with baje
  if (h === 24) h = 0;
  return h * 60 + min;
}

const _PRODUCT_NOUN = '(product|products|dawa|dawai|medicine|item|option|options|kit|combo|pack'

  || new RegExp('\\b(kya|kaun|konsa|kaunsa|kaunse|kis)\\b\\s*(\\S+\\s+){0,2}(chij|cheez|chiz|chize|cheez)\\b', 'i').test(_pq)

  || /\b(recommend|suggest)\b/i.test(_pq)
  || /\bke\s*liye\b[^.!?]{0,24}\b(kya|konsa|konse|kaunsa|product)\b/i.test(_pq);

// ---- SUBSTANCE GUARD ----

    : (_weAskedQuestion ? 'answered_our_question' : 'unanswered_question'));

    ? 'Ji, note kar liya. Aap apna naam aur kaunsa time aapke liye theek rahega bata dijiye ('

    // gap - but do not assume the question was medical either. "Recording karke bataiye kahan
    // se aap" is a question about the COMPANY, and answering it with "doctor aapko samjha
    // denge" is a non-sequitur that reads as the bot not having understood at all.
    : _healthOrProductQ

// ---- REPEAT GUARD ----
// "ji yah drops hai" went out VERBATIM twice in a row. Nothing has ever compared an outgoing
// reply with the previous one; the only defence was CRITICAL ANTI-SPAM rule 5 in the prompt,
// and this model does not obey prompt rules (which is why every other guard here exists).
// Repeating yourself is the single most obvious way to look like a broken bot, and it is
// what the customer reacts to by repeating THEMSELVES - which is how these loops lengthen.
// Guard-built replies are exempt: they are correct answers, and if the customer asks the same
// question twice the same price or dose answer is the right one to send again.
let repeatGuard = false;
const _normReply = (s) => String(s || '').toLowerCase().replace(/[^a-z0-9 ]+/g, ' ').replace(/\s+/g, ' ').trim();
if (!_answered() && !doseGuard && !slotGuard && !bookingGuard && !priceGuard && !promiseGuard && !substanceGuard && !pitchGuard) {
  const _now = _normReply(aiText), _prev = _normReply(_lastOutText);
  if (_now && _prev && (_now === _prev || (_now.length >= 12 && (_prev.includes(_now) || _now.includes(_prev))))) {
    repeatGuard = true;
    aiText = 'Ji, lagta hai main aapko theek se samjha nahi paayi. Free consultation me doctor aapke saare sawaal ka jawaab de denge, wo bilkul free hai. '

  }
}

// ---- SAFETY CLAIM GUARD ----
// "jiska koi side effect nhi hota hai" shipped in the same 7 Aug reply. This is a regulatory
// claim about a medicine, not a matter of tone: we cannot promise any product is
// side-effect-free, cures anything, or is 100% safe, and persona Rule 4 forbids exactly this
// ("no fake claims/cure guarantees, no 100% cure").
// Deliberately NOT gated on the guards above - whoever wrote the sentence, model or guard, it
// must not reach a customer - and placed last so it sees the final text.
// Every branch requires an explicit negation or an absolute word, so a customer asking "iska
// side effect hai kya?" still gets a normal answer; only the CLAIM of no side effects trips it.
let safetyClaimGuard = false;
const _SAFETY_CLAIM_RE = new RegExp(
     '\\b(side\\s*effects?|nuksan|nuksaan|haani)\\b[^.!?]{0,24}\\b(nahi|nhi|nai)\\b'
  + '|\\bno\\s+side\\s*effects?\\b'
  + '|\\b100\\s*(%|percent|pratishat)\\s*(safe|sure|cure|guaranteed?|result)\\b'
  + '|\\b(bilkul|puri\\s*tarah|puri\\s*trah|totally|completely)\\s*safe\\b'
  + '|\\b(guarantee|guaranteed)\\b[^.!?]{0,20}\\b(result|cure|ilaj|theek|thik)\\b', 'i');
// Checked against the FINAL text and against the model's own words. If a guard above already
// replaced the claim with something safe we leave that better answer alone and only record the
// attempt - overwriting it would throw away a real price or dose answer to say less.
const _claimInReply = _SAFETY_CLAIM_RE.test(String(aiText));
const _claimByModel = _SAFETY_CLAIM_RE.test(_modelText);
if (_claimInReply || _claimByModel) {
  const _blocked = (_claimInReply ? String(aiText) : _modelText).slice(0, 80);
  safetyClaimGuard = _claimInReply ? 'blocked' : 'model_only';
  if (_claimInReply) {
    aiText = 'Ji, kisi bhi product ki safety ya result ka dava main khud se nahi kar sakti - ye doctor hi aapki '
      + 'puri history dekh kar batate hain. Aap apna naam aur kaunsa time aapke liye theek rahega bata dijiye, '
      + 'main free consultation set kara deti hu.';
  }
  console.log('[Safety Claim Guard] ' + safetyClaimGuard + ' an absolute safety/cure claim | was='
    + JSON.stringify(_blocked));
}
